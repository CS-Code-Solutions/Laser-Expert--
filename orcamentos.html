window.compartilharPDFOrcamento = async function () { const status = document.getElementById('loadingStatusOrc'); const source = document.getElementById('budgetRenderArea'); // Verifica se existe conteúdo para gerar o PDF if ( !source || !source.querySelector('.print-body') || !source.innerText.trim() ) { alert( 'Não foi possível gerar o PDF: o conteúdo do orçamento está vazio. Feche e abra o orçamento novamente.' ); return; } // Verifica se a biblioteca está disponível if (typeof html2pdf === 'undefined') { alert( 'A biblioteca de geração de PDF não foi carregada. Verifique sua conexão e tente novamente.' ); return; } let captureHost; try { status.textContent = 'Preparando orçamento...'; // Cria uma área independente para capturar o conteúdo captureHost = document.createElement('div'); captureHost.style.cssText = [ 'position:fixed', 'left:-10000px', 'top:0', 'width:794px', 'background:#ffffff', 'color:#0B132B', 'opacity:1', 'visibility:visible', 'pointer-events:none', 'z-index:2147483647', 'overflow:visible', 'transform:none' ].join(';'); // Clona o conteúdo do orçamento const clone = source.cloneNode(true); clone.removeAttribute('id'); clone.style.cssText += ';display:block;' + 'width:100%;' + 'height:auto;' + 'min-height:0;' + 'overflow:visible;' + 'background:#ffffff;' + 'transform:none;'; const printBody = clone.querySelector('.print-body'); if (printBody) { printBody.style.width = '100%'; printBody.style.maxWidth = 'none'; printBody.style.color = '#0B132B'; printBody.style.background = '#ffffff'; } captureHost.appendChild(clone); document.body.appendChild(captureHost); // Aguarda o carregamento das fontes if (document.fonts && document.fonts.ready) { await document.fonts.ready; } // Aguarda o carregamento das imagens const images = Array.from( captureHost.querySelectorAll('img') ); await Promise.all( images.map(async (img) => { if (img.complete && img.naturalWidth > 0) { return; } if (typeof img.decode === 'function') { try { await img.decode(); return; } catch (e) { // Continua mesmo se a imagem falhar } } await new Promise((resolve) => { img.onload = resolve; img.onerror = resolve; }); }) ); // Verifica se o conteúdo possui dimensões válidas const rect = clone.getBoundingClientRect(); if ( !rect.width || !rect.height || !clone.querySelector('.print-body') ) { throw new Error( 'O conteúdo do orçamento não possui dimensões válidas para captura.' ); } status.textContent = 'Gerando PDF...'; const numeroOrcamento = (activeBudgetData && activeBudgetData.number) || '001'; const filename = `Orcamento_${numeroOrcamento}.pdf`; // Gera e salva o PDF await html2pdf() .set({ margin: 10, filename: filename, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, backgroundColor: '#ffffff', logging: false, scrollX: 0, scrollY: 0, windowWidth: 794 }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' }, pagebreak: { mode: ['css', 'legacy'] } }) .from(clone) .save(); status.textContent = 'PDF baixado!'; } catch (error) { console.error( 'Erro ao gerar PDF do orçamento:', error ); status.textContent = 'Falha ao gerar PDF'; alert( 'Não foi possível gerar o PDF. Tente novamente. ' + 'Se o problema continuar, abra o console do navegador para verificar o erro.' ); } finally { // Remove a área temporária if (captureHost && captureHost.parentNode) { captureHost.parentNode.removeChild(captureHost); } } };
